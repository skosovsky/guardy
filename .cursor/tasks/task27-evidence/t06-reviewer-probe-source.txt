package main
import("context";"errors";"fmt";g "github.com/skosovsky/guardy";"github.com/skosovsky/guardy/ext";j "github.com/skosovsky/guardy/ext/jsonredact")
func main(){
 for _,which:=range []string{"slice","json"}{
 ctx,cancel:=context.WithCancel(context.Background()); cause:=errors.New("fault")
 leaf:=g.ValidatorFunc[string](func(context.Context,string)(string,*g.Report,error){cancel();return "bad",nil,g.WithCompletedObservations(cause,&g.Report{Action:g.ActionPass,PayloadKind:g.PayloadInternalControlSignal})})
 var err error
 if which=="slice"{_,_,err=ext.MapSlice(func(s string)string{return s},func(_,s string)string{return s},leaf).Validate(ctx,[]string{"x"})}else{_,_,err=j.NewJSONRedactValidator(leaf,"").Validate(ctx,`["x"]`)}
 fmt.Printf("%s fault=%t cancellation=%t kind=%v\n",which,errors.Is(err,cause),errors.Is(err,context.Canceled),g.CompletedReportFromError(err).PayloadKind)
 }
}
