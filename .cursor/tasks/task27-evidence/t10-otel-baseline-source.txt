package guardyotel
import (
 "errors"
 "testing"
 "go.opentelemetry.io/otel/metric"
 "go.opentelemetry.io/otel/metric/noop"
)
type setupFailMeter struct { metric.Meter; counterErr, histogramErr error }
func (m setupFailMeter) Int64Counter(name string, opts ...metric.Int64CounterOption) (metric.Int64Counter,error) {
 if m.counterErr != nil { return nil,m.counterErr }; return m.Meter.Int64Counter(name,opts...)
}
func (m setupFailMeter) Float64Histogram(name string, opts ...metric.Float64HistogramOption) (metric.Float64Histogram,error) {
 if m.histogramErr != nil { return nil,m.histogramErr }; return m.Meter.Float64Histogram(name,opts...)
}
func TestT10MeterSetupErrorCannotBeIgnored(t *testing.T) {
 for _, histogram := range []bool{false,true} {
  // Arrange.
  meter:=setupFailMeter{Meter:noop.NewMeterProvider().Meter("test")}
  sentinel:=errors.New("provider setup failure")
  if histogram { meter.histogramErr=sentinel } else { meter.counterErr=sentinel }
  // Act: baseline has no error channel; it must not admit a usable wrapper on failure.
  middleware:=NewMiddleware[string](WithMeter(meter),WithTracer(nil))
  // Assert.
  if middleware != nil { t.Error("failed instrument setup silently admitted middleware") }
 }
}
