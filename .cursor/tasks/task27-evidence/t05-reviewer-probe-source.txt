package main

import (
 "bytes"
 "context"
 "errors"
 "fmt"
 "io"
 "time"
 g "github.com/skosovsky/guardy"
)

type shortWriter struct { calls int }
func (w *shortWriter) Write(p []byte) (int,error) { w.calls++; return 1,nil }
func cfg(fn g.ValidatorFunc[string]) g.StreamConfig {
 return g.StreamConfig{Identity:"probe", Profile:g.ReleaseValidatedUnits, JSONValues:true,
 Pipeline:g.NewPipeline(g.WithFastPath(g.WithStreamingCapabilities(fn,g.StreamCapabilities{Unit:true}))),
 Delivery:g.NewUserTextPolicy("internal",g.WithDeliveryAllowedKinds(g.PayloadSafeUserText,g.PayloadTechnicalPayload,g.PayloadInternalControlSignal)),
 MaxInputBytes:100,MaxPendingBytes:9,MaxUnitBytes:8,MaxOutputBytes:100,ValidationTimeout:time.Second}
}
func check(ok bool, message string) { if !ok { panic(message) } }
func main() {
 // Arrange: successful prefix, then shape-valid JSON redaction exceeding byte unit budget.
 c:=cfg(func(_ context.Context,s string)(string,*g.Report,error){
  if s==`{"x":1}` { return `{"long":1}`,&g.Report{Action:g.ActionRedact,PayloadKind:g.PayloadInternalControlSignal},nil }; return s,nil,nil
 })
 c.Delivery.Classifier=func(any)(g.PayloadKind,error){return g.PayloadSafeUserText,nil}
 var sink bytes.Buffer
 p,e:=g.CompileStream(&sink,c); check(e==nil,"compile prefix")
 // Act.
 _,e=p.Write([]byte(`{}{"x":1}{}`))
 // Assert.
 fmt.Printf("prefix: err=%v sink=%q outcome=%+v\n",e,sink.String(),p.Outcome())
 check(errors.Is(e,g.ErrStreamUnitLimit)&&sink.String()==`{}`&&p.Outcome().Decision.PayloadKind==g.PayloadInternalControlSignal,"prefix or classified unit budget")
 _,e=p.DeliverFallback(context.Background()); check(e!=nil,"limit activated fallback")

 // Arrange: denied source, cancelled fallback validator returns an apparently approved replacement.
 ctx,cancel:=context.WithCancel(context.Background())
 c=cfg(func(_ context.Context,s string)(string,*g.Report,error){if s==`{}` {return s,&g.Report{Action:g.ActionBlock},nil}; cancel(); return `[]`,nil,nil})
 c.Delivery.Fallback=`[1]`; sink.Reset(); p,e=g.CompileStream(&sink,c); check(e==nil,"compile cancel")
 p.Write([]byte(`{}`)); p.Complete(context.Background())
 // Act.
 o,e:=p.DeliverFallback(ctx)
 // Assert.
 check(errors.Is(e,context.Canceled)&&o.Category==g.StreamTimeout&&sink.Len()==0,"cancelled fallback delivered")

 // Arrange: separately accepted fallback reaches a short writer exactly once.
 c=cfg(func(_ context.Context,s string)(string,*g.Report,error){if s==`{}` {return s,&g.Report{Action:g.ActionBlock},nil}; return s,nil,nil})
 c.Delivery.Fallback=`[]`; w:=new(shortWriter); p,e=g.CompileStream(w,c); check(e==nil,"compile transport")
 p.Write([]byte(`{}`)); p.Complete(context.Background()); original:=p.Outcome()
 // Act.
 o,e=p.DeliverFallback(context.Background()); _,again:=p.DeliverFallback(context.Background())
 // Assert.
 check(errors.Is(e,io.ErrShortWrite)&&o.Category==g.StreamTransport&&o.ReleasedBytes==1&&w.calls==1&&again!=nil&&p.Outcome()==original,"fallback transport sticky accounting")
 fmt.Println("PASS: prefix unit budget and prior classification; limit forbids fallback; cancelled fallback zero writes; fallback short transport at-most-once and original outcome sticky")
}
