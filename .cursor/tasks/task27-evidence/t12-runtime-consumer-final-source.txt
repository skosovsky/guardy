import (
 "bytes"
 "context"
 "errors"
 "net/http"
 "testing"
 "time"
 g "github.com/skosovsky/guardy"
 "github.com/skosovsky/guardy/build"
 "github.com/skosovsky/guardy/ext/guardyotel"
 "github.com/skosovsky/guardy/ext/jsonredact"
 "github.com/skosovsky/guardy/ext/jsonschema"
)
func TestReleasedConsumer(t *testing.T) {
 // Arrange.
 leaf:=g.ValidatorFunc[string](func(context.Context,string)(string,*g.Report,error){return "clean", &g.Report{Action:g.ActionRedact,MutatedText:"wrong mirror"},nil})
 p,err:=g.NewPipeline(g.WithSequential(leaf));if err!=nil{t.Fatal(err)}
 middleware,err:=guardyotel.NewMiddleware[string](guardyotel.WithMeter(nil),guardyotel.WithTracer(nil));if err!=nil{t.Fatal(err)}
 p,err=p.Use(middleware);if err!=nil{t.Fatal(err)}
 // Act.
 delivery,err:=p.GuardOutput(context.Background(),nil,"secret")
 projection,ok:=delivery.Projection()
 // Assert.
 if err!=nil||!ok||projection.Value!="clean"{t.Fatal("authoritative projection failed")}
 // Arrange.
 fail:=g.ValidatorFunc[string](func(_ context.Context,v string)(string,*g.Report,error){return v,&g.Report{Disposition:g.DispositionSystemFault},nil})
 fault,err:=g.NewPipeline(g.WithSequential(fail));if err!=nil{t.Fatal(err)}
 // Act.
 blocked,err:=fault.GuardOutput(context.Background(),nil,"secret")
 _,ok=blocked.Projection()
 // Assert.
 var failure *g.PolicyFailure
 if err==nil||ok||!errors.As(err,&failure)||!failure.Decision.IsSystemFault(){t.Fatal("report-only fault delivered")}
 // Arrange.
 schema,err:=jsonschema.NewJSONSchemaValidator(`{"type":"object","properties":{"n":{"const":9007199254740993}},"required":["n"]}`);if err!=nil{t.Fatal(err)}
 jsonLeaf:=jsonredact.NewJSONRedactValidator(leaf,"redact")
 compiled,err:=build.CompileStringGuard(build.GuardSpec{});if err!=nil{t.Fatal(err)}
 httpGuard,err:=g.Guard(compiled,func(*http.Request)(string,error){return "safe",nil},g.PlainTextInjector(),g.WithGuardMaxBodyBytes(3));if err!=nil||httpGuard==nil{t.Fatal(err)}
 // Act.
 _,report,err:=schema.Validate(context.Background(),`{"n":9007199254740993}`)
 clean,redaction,redactErr:=jsonLeaf.Validate(context.Background(),`{"email":"secret"}`)
 // Assert.
 if err!=nil||g.DecisionFromReport(report).IsTerminal()||redactErr!=nil||clean!=`{"email":"clean"}`||g.DecisionFromReport(redaction).Action!=g.ActionRedact{t.Fatalf("optional contracts failed: %v %s %v",err,clean,redactErr)}
 // Arrange: stream must preserve completed kind and simultaneous causes.
 streamCtx,cancel:=context.WithCancel(context.Background());defer cancel()
 sentinel:=errors.New("independent detector failure")
 caps:=g.StreamCapabilities{Unit:true,Final:true}
 first:=g.WithStreamingCapabilities(g.ValidatorFunc[string](func(_ context.Context,v string)(string,*g.Report,error){return v,&g.Report{PayloadKind:g.PayloadTechnicalPayload},nil}),caps)
 second:=g.WithStreamingCapabilities(g.ValidatorFunc[string](func(_ context.Context,v string)(string,*g.Report,error){cancel();return v,nil,sentinel}),caps)
 streamPipeline,err:=g.NewPipeline(g.WithSequential(first,second));if err!=nil{t.Fatal(err)}
 var sink bytes.Buffer
 stream,err:=g.CompileStream(&sink,g.StreamConfig{Identity:"consumer-stream",Pipeline:streamPipeline,Profile:g.ReleaseValidatedUnits,MaxInputBytes:100,MaxOutputBytes:100,MaxUnitBytes:20,MaxPendingBytes:20,ValidationTimeout:time.Second,Delivery:g.NewUserTextPolicy("internal",g.WithDeliveryAllowedKinds(g.PayloadTechnicalPayload))});if err!=nil{t.Fatal(err)}
 // Act.
 _,err=stream.WriteContext(streamCtx,[]byte("hello\n"))
 outcome:=stream.Outcome()
 // Assert.
 if !errors.Is(err,sentinel)||!errors.Is(err,context.Canceled)||outcome.Decision.PayloadKind!=g.PayloadTechnicalPayload||!outcome.Decision.IsSystemFault()||sink.Len()!=0{t.Fatal("stream evidence projection failed")}
}
''')
