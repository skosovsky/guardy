package guardy
import "testing"
func TestT08NegativeRoutingBaseline(t *testing.T) {
 // Arrange.
 decision := Decision{Action: ActionRetry, Disposition: DispositionRetryableCorrection}
 // Act.
 route := RouteDecision(decision, RemediationPolicy{RetryAttempt: -1, MaxRetries: 2})
 // Assert.
 if route.Outcome == GuardRouteRetryCorrection { t.Fatal("negative attempt admitted as retry") }
}
