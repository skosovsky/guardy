package guardy
import("io";"net/http";"net/http/httptest";"strings";"testing")
type baselineBody struct { io.Reader; closes int }
func(b *baselineBody)Close()error{b.closes++;return nil}
func TestT10HTTPBaselineOwnershipAndLimit(t *testing.T){
 for _,oversized:=range []bool{false,true}{
  // Arrange.
  raw:="ok";if oversized{raw=strings.Repeat("x",DefaultMaxBodyBytes+1)}
  body:=&baselineBody{Reader:strings.NewReader(raw)}
  req:=httptest.NewRequest(http.MethodPost,"/",nil);req.Body=body
  handler:=Guard(MustNewPipeline[string](),bodyExtractor,PlainTextInjector())(http.HandlerFunc(func(http.ResponseWriter,*http.Request){}))
  rec:=httptest.NewRecorder()
  // Act.
  handler.ServeHTTP(rec,req)
  // Assert.
  if body.closes!=1{t.Error("consumed wrapper not closed")}
  if oversized&&rec.Code!=http.StatusRequestEntityTooLarge{t.Errorf("oversized status=%d want413",rec.Code)}
 }
}
